import { ICivicTaxRepository } from './ICivicTaxRepository.js';
import {
  IProperty,
  ITaxAssessment,
  ITaxPayment,
  IGrievance,
  IReassessment,
  IDashboardMetrics,
} from './types.js';
import { SalesforceSessionData } from '../types/index.js';
import { SalesforceClient } from '../services/salesforceClient.js';

type SalesforceRecord = {
  [key: string]: any;
};

export class SalesforceRepository implements ICivicTaxRepository {
  constructor(private readonly session: SalesforceSessionData) {
    if (!session?.accessToken || !session?.instanceUrl) {
      throw new Error('A valid Salesforce session is required.');
    }
  }

  private async call(
    endpoint: string,
    method: 'GET' | 'POST' | 'PATCH' = 'GET',
    body?: unknown
  ): Promise<any> {
    return SalesforceClient.callApexRest(
      this.session,
      endpoint,
      method,
      body
    );
  }

  private unsupported(operation: string): never {
    throw new Error(
      `${operation} is not available in the current Salesforce Apex REST API.`
    );
  }

  private mapProperty(record: SalesforceRecord): IProperty {
    const category = String(
      record.Property_Category__c ?? 'RESIDENTIAL'
    ).toUpperCase();

    const validCategories = [
      'RESIDENTIAL',
      'COMMERCIAL',
      'INDUSTRIAL',
      'AGRICULTURAL',
    ];

    const verificationStatus = String(
      record.Verification_Status__c ?? 'PENDING'
    ).toUpperCase();

    const validStatuses = [
      'PENDING',
      'VERIFIED',
      'RESUBMISSION_REQUIRED',
      'REJECTED',
    ];

    return {
      id: record.Id,
      propertyNumber:
        record.Property_Number__c ?? record.Name ?? '',
      ownerId: record.OwnerId ?? record.Property_Owner__c ?? '',
      ownerName:
        record.Property_Owner__r?.Name ??
        record.Owner?.Name ??
        'Not available',
      address: record.Property_Address__c ?? '',
      category: validCategories.includes(category)
        ? category as IProperty['category']
        : 'RESIDENTIAL',
      builtUpArea: Number(record.Built_Up_Area__c ?? 0),
      zone: record.Zone__c ?? record.Municipal_Zone__c ?? '',
      verificationStatus: validStatuses.includes(verificationStatus)
        ? verificationStatus as IProperty['verificationStatus']
        : 'PENDING',
      documentReference: record.Ownership_Document__c ?? '',
      latitude:
        record.Latitude__c == null
          ? undefined
          : Number(record.Latitude__c),
      longitude:
        record.Longitude__c == null
          ? undefined
          : Number(record.Longitude__c),
      constructionYear:
        record.Construction_Year__c == null
          ? undefined
          : Number(record.Construction_Year__c),
      registrationDate: record.Registration_Date__c ?? '',
    };
  }

  private mapAssessment(
    record: SalesforceRecord,
    property?: IProperty
  ): ITaxAssessment {
    const baseTax = Number(
      record.Base_Tax__c ?? record.Base_Value__c ?? 0
    );

    const totalTax = Number(
      record.Total_Tax__c ??
      record.Total_Amount__c ??
      baseTax
    );

    const amountPaid = Number(record.Amount_Paid__c ?? 0);

    const outstandingBalance = Number(
      record.Outstanding_Amount__c ??
      Math.max(0, totalTax - amountPaid)
    );

    const rawStatus = String(
      record.Assessment_Status__c ?? 'DRAFT'
    ).toUpperCase();

    const validStatuses = [
      'DRAFT',
      'APPROVED',
      'PARTIALLY_PAID',
      'PAID',
      'OVERDUE',
    ];

    return {
      id: record.Id,
      assessmentNumber:
        record.Assessment_Number__c ??
        record.Name ??
        record.Id,
      propertyId: record.Property__c ?? '',
      assessmentYear: record.Assessment_Year__c ?? '',
      category: property?.category ?? '',
      builtUpArea: property?.builtUpArea ?? 0,
      appliedRatePerSqFt: Number(
        record.Applied_Rate_Per_Sq_Ft__c ?? 0
      ),
      baseTax,
      penalty: Number(record.Penalty__c ?? 0),
      discount: Number(record.Discount__c ?? 0),
      totalTax,
      amountPaid,
      outstandingBalance,
      status: validStatuses.includes(rawStatus)
        ? rawStatus as ITaxAssessment['status']
        : 'DRAFT',
      dueDate: record.Due_Date__c ?? '',
    };
  }

  private mapPayment(record: SalesforceRecord): ITaxPayment {
    return {
      id: record.Id,
      assessmentId: record.Assessment__c ?? '',
      amountPaid: Number(record.Amount_Paid__c ?? 0),
      paymentDate: record.Payment_Date__c ?? '',
      paymentMethod: record.Payment_Method__c ?? '',
      transactionId: record.Transaction_ID__c ?? '',
      paymentStatus:
        record.Payment_Status__c === 'SUCCESS' ||
        record.Payment_Status__c === 'FAILED'
          ? record.Payment_Status__c
          : 'PENDING',
      receiptNumber: record.Name ?? record.Id,
    };
  }

  // PROPERTIES

  public async listProperties(
    filter?: { category?: string; zone?: string; status?: string }
  ): Promise<IProperty[]> {
    const records = await this.call('properties');

    if (!Array.isArray(records)) {
      throw new Error('Unexpected Salesforce properties response.');
    }

    let properties = records.map((record) =>
      this.mapProperty(record)
    );

    if (filter?.category) {
      properties = properties.filter(
        (property) => property.category === filter.category
      );
    }

    if (filter?.zone) {
      properties = properties.filter(
        (property) => property.zone === filter.zone
      );
    }

    if (filter?.status) {
      properties = properties.filter(
        (property) =>
          property.verificationStatus === filter.status
      );
    }

    return properties;
  }

  public async getPropertyById(
    id: string
  ): Promise<IProperty | null> {
    try {
      const record = await this.call(
        `properties/${encodeURIComponent(id)}`
      );

      return record?.Id ? this.mapProperty(record) : null;
    } catch (error: any) {
      if (
        error.message?.includes('(404)') ||
        error.message?.includes('NOT_FOUND')
      ) {
        return null;
      }

      throw error;
    }
  }

  public async createProperty(
    property: Omit<IProperty, 'id' | 'registrationDate'>
  ): Promise<IProperty> {
    const record = await this.call('properties', 'POST', {
      propertyNumber: property.propertyNumber,
      address: property.address,
      category: property.category,
      builtUpArea: property.builtUpArea,
      zone: property.zone,
      latitude: property.latitude,
      longitude: property.longitude,
      documentReference: property.documentReference,
      constructionYear: property.constructionYear,
    });

    if (!record?.Id) {
      throw new Error(
        'Salesforce did not return the created property record.'
      );
    }

    return this.mapProperty(record);
  }

  public async updatePropertyStatus(
    id: string,
    status: IProperty['verificationStatus']
  ): Promise<IProperty> {
    const record = await this.call('properties', 'PATCH', {
      propertyId: id,
      status,
      notes: '',
    });

    if (!record?.Id) {
      throw new Error(
        'Salesforce did not return the updated property record.'
      );
    }

    return this.mapProperty(record);
  }

  // ASSESSMENTS

  public async listAssessments(
    propertyId?: string
  ): Promise<ITaxAssessment[]> {
    const [records, properties] = await Promise.all([
      this.call('assessments'),
      this.listProperties(),
    ]);

    if (!Array.isArray(records)) {
      throw new Error('Unexpected Salesforce assessments response.');
    }

    const propertyMap = new Map(
      properties.map((property) => [property.id, property])
    );

    let assessments = records.map((record) =>
      this.mapAssessment(
        record,
        propertyMap.get(record.Property__c)
      )
    );

    if (propertyId) {
      assessments = assessments.filter(
        (assessment) => assessment.propertyId === propertyId
      );
    }

    return assessments;
  }

  public async getAssessmentById(
    id: string
  ): Promise<ITaxAssessment | null> {
    const assessments = await this.listAssessments();

    return (
      assessments.find(
        (assessment) =>
          assessment.id === id ||
          assessment.assessmentNumber === id
      ) ?? null
    );
  }

  public async createAssessment(
    _assessment: Omit<ITaxAssessment, 'id'>
  ): Promise<ITaxAssessment> {
    return this.unsupported('Assessment creation');
  }

  // PAYMENTS

  public async listPayments(
    _assessmentId?: string
  ): Promise<ITaxPayment[]> {
    return this.unsupported('Payment history retrieval');
  }

  public async createPayment(
    payment: Omit<ITaxPayment, 'id' | 'receiptNumber'>
  ): Promise<{
    payment: ITaxPayment;
    assessment: ITaxAssessment;
  }> {
    const currentAssessment = await this.getAssessmentById(
      payment.assessmentId
    );

    if (!currentAssessment) {
      throw new Error('Assessment not found in Salesforce.');
    }

    if (payment.amountPaid <= 0) {
      throw new Error('Payment amount must be greater than zero.');
    }

    if (payment.amountPaid > currentAssessment.outstandingBalance) {
      throw new Error('Payment exceeds the outstanding assessment balance.');
    }

    const record = await this.call('payments', 'POST', {
      assessmentId: payment.assessmentId,
      amountPaid: payment.amountPaid,
      paymentMethod: payment.paymentMethod,
      transactionId: payment.transactionId,
    });

    if (!record?.Id) {
      throw new Error(
        'Salesforce did not return the created payment record.'
      );
    }

    const createdPayment = this.mapPayment(record);

    // Apex PaymentService updates the assessment balance atomically.
    // Reflect the confirmed payment in the returned assessment.
    const updatedAssessment: ITaxAssessment = {
      ...currentAssessment,
      amountPaid:
        Math.round(
          (currentAssessment.amountPaid + createdPayment.amountPaid) *
          100
        ) / 100,
      outstandingBalance:
        Math.round(
          (currentAssessment.outstandingBalance -
            createdPayment.amountPaid) *
          100
        ) / 100,
      status:
        currentAssessment.outstandingBalance -
          createdPayment.amountPaid <= 0
          ? 'PAID'
          : 'PARTIALLY_PAID',
    };

    return {
      payment: createdPayment,
      assessment: updatedAssessment,
    };
  }

  // GRIEVANCES

  public async listGrievances(
    _citizenId?: string
  ): Promise<IGrievance[]> {
    return this.unsupported('Grievance retrieval');
  }

  public async createGrievance(
    _grievance: Omit<
      IGrievance,
      'id' | 'grievanceNumber' | 'createdAt'
    >
  ): Promise<IGrievance> {
    return this.unsupported('Grievance creation');
  }

  public async resolveGrievance(
    _id: string,
    _comments: string
  ): Promise<IGrievance> {
    return this.unsupported('Grievance resolution');
  }

  // REASSESSMENTS

  public async listReassessments(
    _propertyId?: string
  ): Promise<IReassessment[]> {
    return this.unsupported('Reassessment retrieval');
  }

  public async createReassessment(
    _reassessment: Omit<
      IReassessment,
      'id' | 'requestNumber' | 'createdAt'
    >
  ): Promise<IReassessment> {
    return this.unsupported('Reassessment creation');
  }

  // DASHBOARD

  public async getDashboardMetrics(): Promise<IDashboardMetrics> {
    const record = await this.call('dashboard');

    if (!record || typeof record !== 'object') {
      throw new Error('Unexpected Salesforce dashboard response.');
    }

    const totalProperties = Number(record.totalProperties ?? 0);
    const verifiedProperties = Number(
      record.verifiedProperties ?? 0
    );
    const pendingVerifications = Number(
      record.pendingProperties ?? 0
    );
    const totalTaxAssessed = Number(
      record.totalAssessedTax ??
      record.totalTaxAssessed ??
      0
    );
    const totalRevenueCollected = Number(
      record.totalRevenueCollected ?? 0
    );

    const totalOutstandingDues = Math.max(
      0,
      totalTaxAssessed - totalRevenueCollected
    );

    const collectionEfficiencyRate =
      totalTaxAssessed > 0
        ? Math.round(
            (totalRevenueCollected / totalTaxAssessed) * 10000
          ) / 100
        : 0;

    return {
      totalProperties,
      verifiedProperties,
      pendingVerifications,
      totalTaxAssessed,
      totalRevenueCollected,
      totalOutstandingDues,
      collectionEfficiencyRate,

      // The current Apex dashboard does not expose grievance metrics.
      // This is a placeholder, not an authoritative grievance count.
      openGrievances: 0,
    };
  }
}

  public async createPayment(pmt: Omit<ITaxPayment, 'id' | 'receiptNumber'>): Promise<{ payment: ITaxPayment; assessment: ITaxAssessment }> {
    const asm = await this.getAssessmentById(pmt.assessmentId);
    if (!asm) throw new Error('Target assessment not found');

    if (pmt.amountPaid <= 0) throw new Error('Payment amount must be greater than zero');
    if (pmt.amountPaid > asm.outstandingBalance) {
      throw new Error(`Payment amount (₹${pmt.amountPaid}) exceeds remaining balance (₹${asm.outstandingBalance})`);
    }

    const dup = this.payments.find((p) => p.transactionId === pmt.transactionId);
    if (dup) throw new Error(`Duplicate transaction ID: ${pmt.transactionId}`);

    const newPayment: ITaxPayment = {
      ...pmt,
      id: `pay-${Date.now()}`,
      receiptNumber: `RCT-${Date.now()}`,
    };
    this.payments.push(newPayment);

    // Atomically update assessment balance
    asm.amountPaid = Math.round((asm.amountPaid + pmt.amountPaid) * 100) / 100;
    asm.outstandingBalance = Math.round((asm.totalTax - asm.amountPaid) * 100) / 100;
    asm.status = asm.outstandingBalance <= 0 ? 'PAID' : 'PARTIALLY_PAID';

    return { payment: newPayment, assessment: asm };
  }

    grv.resolutionComments = comments;
    grv.resolutionDate = new Date().toISOString();
    return grv;

      createdAt: new Date().toISOString(),
    };
    this.reassessments.push(newReassessment);

  }

  public async getDashboardMetrics(): Promise<IDashboardMetrics> {
    const totalProperties = this.properties.length;
    const verifiedProperties = this.properties.filter((p) => p.verificationStatus === 'VERIFIED').length;
    const pendingVerifications = this.properties.filter((p) => p.verificationStatus === 'PENDING').length;
    
    let totalTaxAssessed = 0;
    let totalRevenueCollected = 0;
    let totalOutstandingDues = 0;

    for (const a of this.assessments) {
      totalTaxAssessed += a.totalTax;
      totalRevenueCollected += a.amountPaid;
      totalOutstandingDues += a.outstandingBalance;
    }

    const collectionEfficiencyRate = totalTaxAssessed > 0 
      ? Math.round((totalRevenueCollected / totalTaxAssessed) * 10000) / 100
      : 0;

    const openGrievances = this.grievances.filter((g) => g.status === 'OPEN').length;

    return {
      totalProperties,
      verifiedProperties,
      pendingVerifications,
      totalTaxAssessed,
      totalRevenueCollected,
      totalOutstandingDues,
      collectionEfficiencyRate,
      openGrievances,
    };
  }
}

    return;

  // Validate state against CSRF
  if (state !== req.session.oauthState) {

  }

  const verifier = req.session.codeVerifier;
  if (!verifier) {
    res.redirect(`${config.frontendUrl}/admin/salesforce?error=Missing_Code_Verifier`);
    return;
  }

  try {
    const tokenData = await SalesforceClient.exchangeCodeForToken(String(code), verifier);

    // Fetch user identity
    let username = '';
    let displayName = '';
    let orgId = '';
    let userId = '';

    try {
      const identity = await SalesforceClient.fetchIdentity(tokenData.id, tokenData.access_token);
      username = identity.username;
      displayName = identity.display_name;
      orgId = identity.organization_id;
      userId = identity.user_id;
    } catch {
      // Identity URL parsing fallback
      const segments = tokenData.id.split('/');
      orgId = segments[segments.length - 2] || '';
      userId = segments[segments.length - 1] || '';
    }

    const sessionData: SalesforceSessionData = {
      accessToken: tokenData.access_token,
      refreshToken: tokenData.refresh_token,
      instanceUrl: tokenData.instance_url,
      identityUrl: tokenData.id,
      userId,
      orgId,
      username,
      displayName,
      connectedAt: new Date().toISOString(),
      lastActiveAt: new Date().toISOString(),
    };

    delete req.session.oauthState;
    delete req.session.codeVerifier;

    res.redirect(`${config.frontendUrl}/admin/salesforce?status=success`);

import { config } from '../config/index.js';
import { SalesforceOAuthTokenResponse, SalesforceSessionData } from '../types/index.js';

export class SalesforceClient {
  // Exchange PKCE Authorization Code for Access and Refresh Tokens
  public static async exchangeCodeForToken(
    code: string,
    codeVerifier: string
  ): Promise<SalesforceOAuthTokenResponse> {
    const tokenEndpoint = `${config.salesforce.loginUrl}/services/oauth2/token`;

    const bodyParams = new URLSearchParams({
      grant_type: 'authorization_code',
      code: code,
      client_id: config.salesforce.clientId,
      redirect_uri: config.salesforce.redirectUri,
      code_verifier: codeVerifier,
    });

    if (config.salesforce.clientSecret) {
      bodyParams.append('client_secret', config.salesforce.clientSecret);
    }

    const response = await fetch(tokenEndpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
        Accept: 'application/json',
      },
      body: bodyParams.toString(),
    });

    if (!response.ok) {
      const errText = await response.text();
      throw new Error(`Salesforce Token Exchange Failed (${response.status}): ${errText}`);
    }

    const tokenData = (await response.json()) as SalesforceOAuthTokenResponse;
    return tokenData;
  }

  // Refresh expired Access Token using Refresh Token
  public static async refreshToken(refreshToken: string): Promise<SalesforceOAuthTokenResponse> {
    const tokenEndpoint = `${config.salesforce.loginUrl}/services/oauth2/token`;

    const bodyParams = new URLSearchParams({
      grant_type: 'refresh_token',
      refresh_token: refreshToken,
      client_id: config.salesforce.clientId,
    });

    if (config.salesforce.clientSecret) {
      bodyParams.append('client_secret', config.salesforce.clientSecret);
    }

    const response = await fetch(tokenEndpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
        Accept: 'application/json',
      },
      body: bodyParams.toString(),
    });

    if (!response.ok) {
      const errText = await response.text();
      throw new Error(`Salesforce Token Refresh Failed (${response.status}): ${errText}`);
    }

    return (await response.json()) as SalesforceOAuthTokenResponse;
  }

  // Fetch Salesforce User Identity (User ID, Org ID, Username, Display Name)
  public static async fetchIdentity(identityUrl: string, accessToken: string) {
    const response = await fetch(identityUrl, {
      headers: {
        Authorization: `Bearer ${accessToken}`,
        Accept: 'application/json',
      },
    });

    if (!response.ok) {
      throw new Error(`Failed to fetch Salesforce identity (${response.status})`);

    method: 'GET' | 'POST' | 'PATCH' | 'DELETE' = 'GET',
    body?: any
  ): Promise<any> {

    const url = `${session.instanceUrl}/services/apexrest/civictax/v1/${cleanPath}`;

    const headers: Record<string, string> = {
      Authorization: `Bearer ${session.accessToken}`,
      Accept: 'application/json',
    };

    if (body && (method === 'POST' || method === 'PATCH')) {
      headers['Content-Type'] = 'application/json';
    }

    const response = await fetch(url, {
  method,
  headers,
  ...(body && ['POST', 'PATCH'].includes(method)
    ? { body: JSON.stringify(body) }
    : {}),
});

    if (response.status === 401 && session.refreshToken) {
      // Refresh token and retry once
      const refreshed = await this.refreshToken(session.refreshToken);
      session.accessToken = refreshed.access_token;
      headers['Authorization'] = `Bearer ${session.accessToken}`;

      const retryRes = await fetch(url, {
  method,
  headers,
  ...(body && ['POST', 'PATCH'].includes(method)
    ? { body: JSON.stringify(body) }
    : {}),
});

      if (!retryRes.ok) {
        const err = await retryRes.text();
        throw new Error(`Apex REST Error after token refresh (${retryRes.status}): ${err}`);
      }
      return await retryRes.json();
    }

    if (!response.ok) {
      const err = await response.text();
      throw new Error(`Apex REST API Error (${response.status}): ${err}`);
    }

    return await response.json();
  }

                    payload.get('longitude') != null ? Decimal.valueOf(String.valueOf(payload.get('longitude'))) : null,
                    (String) payload.get('documentReference'),
                    payload.get('constructionYear') != null ? Integer.valueOf(payload.get('constructionYear')) : null
                );
                List<Property__c> sanitized = (List<Property__c>) CivicTaxSecurityService.sanitizeRecords(AccessType.READABLE, new List<Property__c>{prop});
                res.statusCode = 201;
                res.responseBody = Blob.valueOf(JSON.serialize(sanitized[0]));
            } else if (endpoint.startsWith('payments')) {
                Map<String, Object> payload = (Map<String, Object>) JSON.deserializeUntyped(bodyStr);
                String assessmentId = (String) payload.get('assessmentId');

                // BOLA / IDOR Protection: Validate access to assessment before recording payment
                if (String.isNotBlank(assessmentId)) {
                    CivicTaxSecurityService.requireCanAccessAssessment((Id) assessmentId);
                }

                Tax_Payment__c pmt = PaymentService.recordPayment(
                    assessmentId,
                    Decimal.valueOf(String.valueOf(payload.get('amountPaid'))),
                    (String) payload.get('paymentMethod'),
                    (String) payload.get('transactionId')
                );
                List<Tax_Payment__c> sanitized = (List<Tax_Payment__c>) CivicTaxSecurityService.sanitizeRecords(AccessType.READABLE, new List<Tax_Payment__c>{pmt});
                res.statusCode = 201;
                res.responseBody = Blob.valueOf(JSON.serialize(sanitized[0]));
            } else {
                res.statusCode = 404;
                res.responseBody = Blob.valueOf('{"error":"ENDPOINT_NOT_FOUND"}');
            }
        } catch (CivicTaxSecurityService.SecurityException sex) {
            res.statusCode = 403;
            res.responseBody = Blob.valueOf(JSON.serialize(new Map<String, String>{
                'error' => 'FORBIDDEN',
                'message' => sex.getMessage()
            }));
        } catch (Exception ex) {
            res.statusCode = 400;
            res.responseBody = Blob.valueOf(JSON.serialize(new Map<String, String>{
                'error' => 'BAD_REQUEST',
                'message' => ex.getMessage()
            }));
        }
    }

    // PATCH /services/apexrest/civictax/v1/*
    @HttpPatch
    global static void handlePatch() {
        RestRequest req = RestContext.request;
        RestResponse res = RestContext.response;
        res.addHeader('Content-Type', 'application/json');

        String uri = req.requestURI;
        String endpoint = uri.substringAfter('/civictax/v1/');
        String bodyStr = req.requestBody != null ? req.requestBody.toString() : '';

        try {
            if (endpoint.startsWith('properties')) {
                // Operation-level authorization: Property verification restricted to Officer or Admin
                if (!CivicTaxSecurityService.canVerifyProperty()) {
                    res.statusCode = 403;
                    res.responseBody = Blob.valueOf(JSON.serialize(new Map<String, String>{
                        'error' => 'FORBIDDEN',
                        'message' => 'Property verification is restricted to Municipal Officers and Administrators.'
                    }));
                    return;
                }

                Map<String, Object> payload = (Map<String, Object>) JSON.deserializeUntyped(bodyStr);
                String propertyId = (String) payload.get('propertyId');
                String status = (String) payload.get('status');
                String notes = (String) payload.get('notes');
                Property__c prop = PropertyService.updateVerificationStatus(propertyId, status, notes);
                List<Property__c> sanitized = (List<Property__c>) CivicTaxSecurityService.sanitizeRecords(AccessType.READABLE, new List<Property__c>{prop});

import {
  Property,
  PropertyCategory,
  PropertyFilterOptions,
  PropertyVerificationStatus,
} from '../types';
import { IPropertyService } from './interfaces';
import { BACKEND_BASE } from './salesforceApiClient';

interface BackendProperty {
  id: string;
  propertyNumber?: string;
  ownerId?: string;
  ownerName?: string;
  address?: string;
  category?: string;
  builtUpArea?: number;
  zone?: string;
  verificationStatus?: string;
  documentReference?: string;
  latitude?: number;
  longitude?: number;
  constructionYear?: number;
  registrationDate?: string;
}

export class SalesforcePropertyService implements IPropertyService {
  private async request<T>(path: string, options: RequestInit = {}): Promise<T> {
    const response = await fetch(`${BACKEND_BASE}${path}`, {
      ...options,
      credentials: 'include',
      headers: {
        'Content-Type': 'application/json',
        ...options.headers,
      },
    });

    const result = await response.json().catch(() => ({}));

    if (!response.ok || result.success === false) {
      throw new Error(
        result.error || result.message || `CivicTax API error (${response.status})`
      );
    }

    return result as T;
  }

  private mapProperty(record: BackendProperty): Property {
    const category = ['RESIDENTIAL', 'COMMERCIAL', 'INDUSTRIAL'].includes(
      record.category || ''
    )
      ? (record.category as PropertyCategory)
      : 'RESIDENTIAL';

    const validStatuses: PropertyVerificationStatus[] = [
      'DRAFT',
      'SUBMITTED',
      'PENDING',
      'VERIFIED',
      'REJECTED',
      'RESUBMISSION_REQUIRED',
    ];

    const status = validStatuses.includes(
      record.verificationStatus as PropertyVerificationStatus
    )
      ? (record.verificationStatus as PropertyVerificationStatus)
      : 'PENDING';

    const zone = record.zone || '';

    return {
      id: record.id,
      propertyNumber: record.propertyNumber || '',
      ownerId: record.ownerId || '',
      ownerName: record.ownerName || 'Not available',
      ownerEmail: '',
      ownerPhone: '',
      address: record.address || '',
      category,
      zoneId: zone,
      zoneName: zone,
      builtUpArea: Number(record.builtUpArea || 0),
      constructionYear: Number(record.constructionYear || 0),
      documentReference: record.documentReference || '',
      contactDetails: '',
      description: '',
      verificationStatus: status,
      registrationDate: record.registrationDate || '',
      updatedAt: record.registrationDate || '',
      latitude: record.latitude,
      longitude: record.longitude,
      revisions: [],
    };
  }

  public async getProperties(filter?: PropertyFilterOptions): Promise<Property[]> {
    const result = await this.request<{ data: BackendProperty[] }>(
      '/api/properties'
    );

    let properties = (result.data || []).map((record) =>
      this.mapProperty(record)
    );

    if (filter?.search) {
      const query = filter.search.toLowerCase().trim();
      properties = properties.filter((property) =>
        [
          property.propertyNumber,
          property.ownerName,
          property.address,
          property.documentReference,
        ].some((value) => value.toLowerCase().includes(query))
      );
    }

    if (filter?.category && filter.category !== 'ALL') {
      properties = properties.filter(
        (property) => property.category === filter.category
      );
    }

    if (filter?.zoneId && filter.zoneId !== 'ALL') {
      properties = properties.filter(
        (property) => property.zoneId === filter.zoneId
      );
    }

    if (filter?.verificationStatus && filter.verificationStatus !== 'ALL') {
      properties = properties.filter(
        (property) =>
          property.verificationStatus === filter.verificationStatus
      );
    }

    if (filter?.ownerId) {
      properties = properties.filter(
        (property) => property.ownerId === filter.ownerId
      );
    }

    return properties;
  }

  public async getPropertyById(id: string): Promise<Property | null> {
    try {
      const result = await this.request<{ data: BackendProperty }>(
        `/api/properties/${encodeURIComponent(id)}`
      );
      return result.data ? this.mapProperty(result.data) : null;
    } catch (error: any) {
      if (error.message?.includes('(404)')) return null;
      throw error;
    }
  }

  public async getPropertyByNumber(propertyNumber: string): Promise<Property | null> {
    const properties = await this.getProperties();
    return (
      properties.find(
        (property) =>
          property.propertyNumber.toUpperCase() === propertyNumber.toUpperCase()
      ) || null
    );
  }

 
  public async createProperty(
    data: Omit<
      Property,
      | 'id'
      | 'propertyNumber'
      | 'registrationDate'
      | 'updatedAt'
      | 'verificationStatus'
    >
  ): Promise<Property> {
    const result = await this.request<{ data: BackendProperty }>(
      '/api/properties',
      {
        method: 'POST',
        body: JSON.stringify({
          ownerId: data.ownerId,
          ownerName: data.ownerName,
          address: data.address,
          category: data.category,
          zone: data.zoneName || data.zoneId,
          builtUpArea: data.builtUpArea,
          documentReference: data.documentReference,
          constructionYear: data.constructionYear,
        }),
      }
    );

    if (!result.data) {
      throw new Error(
        'Salesforce did not return the registered property.'
      );
    }

    return this.mapProperty(result.data);
  }

  public async updateProperty(
    id: string,
    data: Partial<Property>
  ): Promise<Property> {
    const result = await this.request<{ data: BackendProperty }>(
      `/api/properties/${encodeURIComponent(id)}`,
      {
        method: 'PATCH',
        body: JSON.stringify(data),
      }
    );

    return this.mapProperty(result.data);
  }

  public async verifyProperty(
    id: string,
    status: PropertyVerificationStatus,
    _officerId: string,
    _officerName: string,
    comments?: string
  ): Promise<Property> {
    const result = await this.request<{ data: BackendProperty }>(
      `/api/properties/${encodeURIComponent(id)}/verify`,
      {
        method: 'PATCH',
        body: JSON.stringify({ status, comments }),
      }
    );

    return this.mapProperty(result.data);
  }

  public async deleteProperty(_id: string): Promise<boolean> {
    throw new Error(
      'Salesforce property deletion is disabled. No Salesforce records have been deleted.'
    );
  }
}

      `CivicTax Property Data Source: ${useSalesforce ? 'Salesforce' : 'Local Demo'}`

  }

  public static getInstance(): ServiceFactory {
    if (!ServiceFactory.instance) {
