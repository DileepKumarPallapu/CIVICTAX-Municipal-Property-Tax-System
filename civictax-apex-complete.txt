public with sharing class PaymentService {
    public class PaymentException extends Exception {}

    // Process a tax payment, update assessment balances and statuses atomically
    public static Tax_Payment__c recordPayment(
        String assessmentId,
        Decimal amount,
        String paymentMethod,
        String transactionId
    ) {
        if (!CivicTaxSecurityService.canCreateObject('Tax_Payment__c')) {
            throw new CivicTaxSecurityService.SecurityException('Access Denied: Insufficient permissions to create Payment records.');
        }

        if (amount == null || amount <= 0) {
            throw new PaymentException('Payment amount must be greater than zero.');
        }
        if (String.isBlank(transactionId)) {
            throw new PaymentException('Transaction reference ID is mandatory.');
        }

        // BOLA / IDOR Protection: Validate caller has access to the target assessment
        if (String.isNotBlank(assessmentId)) {
            CivicTaxSecurityService.requireCanAccessAssessment((Id) assessmentId);
        }

        // Duplicate payment transaction protection
        List<Tax_Payment__c> duplicateCheck = [
            SELECT Id, Transaction_ID__c 
            FROM Tax_Payment__c 
            WHERE Transaction_ID__c = :transactionId 
            LIMIT 1
        ];
        if (!duplicateCheck.isEmpty()) {
            throw new PaymentException('Duplicate transaction reference detected: ' + transactionId);
        }

        // Lock assessment for balance calculation
        List<Tax_Assessment__c> assessments = [
            SELECT Id, Name, Total_Amount__c, Amount_Paid__c, Outstanding_Amount__c, Assessment_Status__c 
            FROM Tax_Assessment__c 
            WHERE Id = :assessmentId 
            FOR UPDATE
        ];
        if (assessments.isEmpty()) {
            throw new PaymentException('Target assessment record not found.');
        }

        Tax_Assessment__c asm = assessments[0];
        Decimal currentPaid = asm.Amount_Paid__c == null ? 0 : asm.Amount_Paid__c;
        Decimal outstanding = asm.Outstanding_Amount__c == null ? asm.Total_Amount__c : asm.Outstanding_Amount__c;

        if (amount > outstanding) {
            throw new PaymentException('Payment amount (â‚¹' + amount + ') exceeds remaining balance (â‚¹' + outstanding + ').');
        }

        // Create Payment record
        Tax_Payment__c payment = new Tax_Payment__c(
            Name = 'PAY-' + System.currentTimeMillis(),
            Assessment__c = asm.Id,
            Amount_Paid__c = amount,
            Payment_Date__c = System.now(),
            Payment_Method__c = paymentMethod,
            Transaction_ID__c = transactionId,
            Payment_Status__c = 'SUCCESS'
        );
        insert payment;

        // Recalculate Assessment
        Decimal newPaid = (currentPaid + amount).setScale(2, System.RoundingMode.HALF_UP);
        Decimal newOutstanding = (asm.Total_Amount__c - newPaid).setScale(2, System.RoundingMode.HALF_UP);

        asm.Amount_Paid__c = newPaid;
        asm.Outstanding_Amount__c = newOutstanding;
        if (newOutstanding <= 0) {
            asm.Assessment_Status__c = 'PAID';
        } else {
            asm.Assessment_Status__c = 'PARTIALLY_PAID';
        }

        update asm;
        return payment;
    }

 * @description Enterprise Service for Tax Payment Processing, Allocation, and Ledger Reconciliation
 * @group CivicTax 6.0 & 10.0

public with sharing class CivicTaxPaymentService {

    public class PaymentException extends Exception {}

    /**
     * @description Processes and records an incoming taxpayer settlement with concurrency and authorization protection
     */
    public static Tax_Payment__c processPayment(Id assessmentId, Decimal amount, String paymentMethod, String transactionId) {
        if (!CivicTaxSecurityService.canCreateObject('Tax_Payment__c')) {
            throw new CivicTaxSecurityService.SecurityException('Access Denied: Insufficient permissions to create Payment records.');
        }

        if (assessmentId == null) {
            throw new PaymentException('Assessment ID is required for payment processing');
        }
        if (amount == null || amount <= 0) {
            throw new PaymentException('Payment amount must be strictly greater than zero');
        }
        if (String.isBlank(transactionId)) {
            throw new PaymentException('Gateway Transaction ID is mandatory');
        }

        // BOLA / IDOR Protection: Validate caller has access to the target assessment
        CivicTaxSecurityService.requireCanAccessAssessment(assessmentId);

        // Duplicate Check
        List<Tax_Payment__c> duplicatePayments = [
            SELECT Id, Transaction_ID__c, Payment_Reference__c 
            FROM Tax_Payment__c 
            WHERE Transaction_ID__c = :transactionId 
            LIMIT 1
        ];
        if (!duplicatePayments.isEmpty()) {
            throw new PaymentException('Duplicate payment detected: Transaction ' + transactionId + ' has already been processed.');
        }

        // Retrieve Assessment with Lock for financial concurrency safety
        List<Tax_Assessment__c> assessments = [
            SELECT Id, Total_Tax__c, Total_Amount__c, Amount_Paid__c, Outstanding_Amount__c, Assessment_Status__c 
            FROM Tax_Assessment__c 
            WHERE Id = :assessmentId 
            FOR UPDATE
        ];

        if (assessments.isEmpty()) {
            throw new PaymentException('Target Assessment record not found');
        }

        Tax_Assessment__c asm = assessments[0];
        Decimal currentOutstanding = asm.Outstanding_Amount__c != null 
            ? asm.Outstanding_Amount__c 
            : (asm.Total_Amount__c != null ? asm.Total_Amount__c : 0.0);

        if (amount > currentOutstanding) {
            throw new PaymentException('Payment amount cannot exceed total outstanding dues of ' + currentOutstanding);
        }

        // Create Payment Record
        Tax_Payment__c pmt = new Tax_Payment__c();
        pmt.Assessment__c = asm.Id;
        pmt.Amount__c = amount;
        pmt.Amount_Paid__c = amount;
        pmt.Payment_Date__c = System.now();
        pmt.Payment_Method__c = paymentMethod;
        pmt.Transaction_ID__c = transactionId;
        pmt.Transaction_Status__c = 'Success';
        pmt.Payment_Status__c = 'SUCCESS';
        pmt.Payment_Reference__c = 'RCT-' + System.currentTimeMillis();

        insert pmt;

        // Recalculate Assessment Balances
        Decimal currentPaid = asm.Amount_Paid__c != null ? asm.Amount_Paid__c : 0.0;
        Decimal newPaid = currentPaid + amount;
        Decimal newOutstanding = currentOutstanding - amount;

        asm.Amount_Paid__c = newPaid;
        asm.Outstanding_Amount__c = newOutstanding;
        asm.Assessment_Status__c = newOutstanding <= 0 ? 'PAID' : 'PARTIALLY_PAID';

        update asm;

        return pmt;
    }

    /**
     * @description Lists all payment receipts for an assessment with access control
     */
    public static List<Tax_Payment__c> getPaymentsForAssessment(Id assessmentId) {
        if (assessmentId == null) {
            return new List<Tax_Payment__c>();
        }

        // Enforce Assessment access
        CivicTaxSecurityService.requireCanAccessAssessment(assessmentId);

        List<Tax_Payment__c> pmts = [
            SELECT Id, Name, Payment_Reference__c, Assessment__c, Amount__c, Amount_Paid__c, 
                   Payment_Date__c, Payment_Method__c, Transaction_Status__c, Transaction_ID__c 
            FROM Tax_Payment__c 
            WHERE Assessment__c = :assessmentId 
            ORDER BY Payment_Date__c DESC
        ];

        return (List<Tax_Payment__c>) CivicTaxSecurityService.sanitizeRecords(AccessType.READABLE, pmts);

/**
 * @description Enterprise Service for Municipal Tax Assessment Calculations & Lifecycle Management
 * @group CivicTax 6.0
 */
public with sharing class CivicTaxAssessmentService {

    public class AssessmentException extends Exception {}

    private static final Map<String, Decimal> BASE_RATES = new Map<String, Decimal>{
        'RESIDENTIAL' => 10.0,
        'COMMERCIAL' => 18.0,
        'INDUSTRIAL' => 25.0,
        'AGRICULTURAL' => 5.0,
        'Residential' => 10.0,
        'Commercial' => 18.0,
        'Industrial' => 25.0,
        'Agricultural' => 5.0
    };

    /**
     * @description Calculates and creates a statutory tax assessment for a verified property
     */
    public static Tax_Assessment__c calculateAndCreateAssessment(Id propertyId, String assessmentYear) {
        if (propertyId == null) {
            throw new AssessmentException('Property ID is mandatory for assessment');
        }

        CivicTaxSecurityService.requireCanManageAssessments();

        List<Property__c> properties = [
            SELECT Id, Property_Number__c, Property_Category__c, Built_Up_Area__c, Verification_Status__c 
            FROM Property__c 
            WHERE Id = :propertyId 
            LIMIT 1
        ];

        if (properties.isEmpty()) {
            throw new AssessmentException('Property not found');
        }

        Property__c prop = properties[0];
        if (prop.Verification_Status__c != 'VERIFIED' && prop.Verification_Status__c != 'Verified') {
            throw new AssessmentException('Tax assessment requires property to be strictly Verified.');
        }

        Decimal rate = BASE_RATES.containsKey(prop.Property_Category__c) 
            ? BASE_RATES.get(prop.Property_Category__c) 
            : 10.0;
        
        Decimal builtArea = prop.Built_Up_Area__c != null ? prop.Built_Up_Area__c : 0.0;
        Decimal baseTax = builtArea * rate;
        Decimal discount = baseTax * 0.05; // 5% prompt payment discount
        Decimal penalty = 0.0;
        Decimal totalTax = baseTax - discount;

        Tax_Assessment__c asm = new Tax_Assessment__c();
        asm.Property__c = prop.Id;
        asm.Assessment_Number__c = 'ASM-' + prop.Property_Number__c + '-' + assessmentYear.replace('-', '');
        asm.Assessment_Year__c = assessmentYear;
        asm.Base_Value__c = baseTax;
        asm.Base_Tax__c = baseTax;
        asm.Discount__c = discount;
        asm.Penalty__c = penalty;
        asm.Total_Tax__c = totalTax;
        asm.Total_Amount__c = totalTax;
        asm.Outstanding_Amount__c = totalTax;
        asm.Amount_Paid__c = 0.0;
        asm.Due_Date__c = Date.today().addDays(60);
        asm.Assessment_Status__c = 'APPROVED';

        insert asm;
        return asm;
    }

    /**
     * @description Fetches all assessments for a property with access control and sanitization
     */
    public static List<Tax_Assessment__c> getAssessmentsForProperty(Id propertyId) {
        if (propertyId == null) {
            return new List<Tax_Assessment__c>();
        }

        // BOLA / IDOR Protection: Validate caller has access to the target property
        CivicTaxSecurityService.requireCanAccessProperty(propertyId);

        List<Tax_Assessment__c> asms = [
            SELECT Id, Name, Assessment_Number__c, Property__c, Assessment_Year__c, 
                   Base_Tax__c, Base_Value__c, Penalty__c, Discount__c, Total_Tax__c, Total_Amount__c, 
                   Amount_Paid__c, Outstanding_Amount__c, Due_Date__c, Assessment_Status__c 
            FROM Tax_Assessment__c 
            WHERE Property__c = :propertyId 
            ORDER BY CreatedDate DESC
        ];

        return (List<Tax_Assessment__c>) CivicTaxSecurityService.sanitizeRecords(AccessType.READABLE, asms);
    }

    public class GrievanceException extends Exception {}

    /**
     * @description Creates a new grievance docket with ownership validation and CRUD enforcement
     */
    public static Grievance__c submitGrievance(Id propertyId, String subject, String description, String priority) {
        if (!CivicTaxSecurityService.canCreateObject('Grievance__c')) {
            throw new CivicTaxSecurityService.SecurityException('Access Denied: Insufficient permissions to create Grievance records.');
        }

        if (String.isBlank(subject) || String.isBlank(description)) {
            throw new GrievanceException('Subject and description are mandatory for grievance filing.');
        }

        // BOLA / IDOR Protection: Citizens can only file grievances against properties they own or have access to
        if (propertyId != null) {
            CivicTaxSecurityService.requireCanAccessProperty(propertyId);
        }

        Grievance__c grv = new Grievance__c();
        grv.Property__c = propertyId;
        grv.Subject__c = subject;
        grv.Description__c = description;
        grv.Priority__c = String.isNotBlank(priority) ? priority.toUpperCase() : 'MEDIUM';
        grv.Status__c = 'OPEN';

        insert grv;
        return grv;
    }

    /**
     * @description Resolves an open grievance with required officer resolution comments
     */
    public static Grievance__c resolveGrievance(Id grievanceId, String resolutionComments) {
        if (grievanceId == null) {
            throw new GrievanceException('Grievance ID cannot be null');
        }
        if (String.isBlank(resolutionComments)) {
            throw new GrievanceException('Mandatory resolution comments must be provided.');
        }

        // Enforce Grievance Officer or Administrator authorization
        CivicTaxSecurityService.requireCanResolveGrievance();

        if (!CivicTaxSecurityService.canUpdateObject('Grievance__c')) {
            throw new CivicTaxSecurityService.SecurityException('Access Denied: Insufficient permissions to update Grievance records.');
        }

        return new SystemElevatedContext().executeResolveGrievance(grievanceId, resolutionComments);
    }

    /**
     * @description Dedicated elevated execution context for authorized municipal officer resolution
     */
    private without sharing class SystemElevatedContext {

                SELECT Id, Status__c, Resolution_Comments__c 
                FROM Grievance__c 
                WHERE Id = :grievanceId 
                LIMIT 1
            ];

            if (grievances.isEmpty()) {
                throw new GrievanceException('Grievance not found');
            }

            Grievance__c grv = grievances[0];
            grv.Status__c = 'RESOLVED';
            grv.Resolution_Comments__c = resolutionComments;

            update grv;
            return grv;
        }

/**
 * CivicTax Security and Authorization Service
 *
 * Citizen identity is resolved through:
 * CivicTax_User__c.CivicTax_ID__c
 * CivicTax_User__c.Salesforce_Contact_ID__c
 *
 * IMPORTANT:
 * The backend must set the authenticated CivicTax ID for every
 * Apex REST transaction using setRequestIdentity().
 */
public with sharing class CivicTaxSecurityService {

    public class SecurityException extends Exception {}

    // Transaction-scoped identity supplied by the authenticated backend.
    private static String requestCivicTaxId;

    /**
     * Set the authenticated CivicTax ID.
     * Call this at the beginning of every Apex REST request.
     * Never take this value directly from an untrusted browser request.
     */
    public static void setRequestIdentity(String civicTaxId) {
        if (String.isBlank(civicTaxId)) {
            throw new SecurityException(
                'Authenticated CivicTax identity is required.'
            );
        }

        requestCivicTaxId = civicTaxId.trim();
    }

    public static String getRequestIdentity() {
        return requestCivicTaxId;
    }

    /**
     * Resolve the CivicTax user record.
     */
    public static CivicTax_User__c getCurrentCivicTaxUser() {

        if (String.isBlank(requestCivicTaxId)) {
            throw new SecurityException(
                'CivicTax identity has not been initialized.'
            );
        }

        List<CivicTax_User__c> users = [
            SELECT Id,
                   CivicTax_ID__c,
                   Salesforce_Contact_ID__c,
                   Full_Name__c,
                   Email__c,
                   Role__c,
                   Account_Status__c
            FROM CivicTax_User__c
            WHERE CivicTax_ID__c = :requestCivicTaxId
            LIMIT 2
        ];

        if (users.isEmpty()) {
            throw new SecurityException(
                'No CivicTax user mapping exists for this account.'
            );
        }

        if (users.size() > 1) {
            throw new SecurityException(
                'Duplicate CivicTax user mappings detected.'
            );
        }

        CivicTax_User__c currentUser = users[0];

        if (currentUser.Account_Status__c != 'ACTIVE') {
            throw new SecurityException(
                'Your CivicTax account is not active.'
            );
        }

        return currentUser;
    }

    /**
     * Resolve the Salesforce Contact belonging to the CivicTax user.
     */
    public static Id getCurrentCitizenContactId() {

        CivicTax_User__c currentUser = getCurrentCivicTaxUser();

        String contactIdText =
            currentUser.Salesforce_Contact_ID__c;

        if (String.isBlank(contactIdText)) {
            throw new SecurityException(
                'Your CivicTax account is not linked to a Salesforce Contact.'
            );
        }

        Id contactId;

        try {
            contactId = Id.valueOf(contactIdText.trim());
        } catch (Exception ex) {
            throw new SecurityException(
                'The Salesforce Contact mapping is invalid.'
            );
        }

        if (contactId.getSObjectType() != Contact.SObjectType) {
            throw new SecurityException(
                'The mapped record is not a Salesforce Contact.'
            );
        }

        List<Contact> contacts = [
            SELECT Id
            FROM Contact
            WHERE Id = :contactId
            LIMIT 1
        ];

        if (contacts.isEmpty()) {
            throw new SecurityException(
                'The mapped Salesforce Contact does not exist.'
            );
        }

        return contactId;
    }

    // ---------------------------------------------------------
    // PROPERTY OWNERSHIP
    // ---------------------------------------------------------

    public static Boolean canAccessProperty(Id propertyId) {

        if (propertyId == null) {
            return false;
        }

        if (isOfficerOrAdmin()) {
            return true;
        }

        Id citizenContactId = getCurrentCitizenContactId();

        List<Property__c> properties = [
            SELECT Id, Owner__c, Property_Owner__c
            FROM Property__c
            WHERE Id = :propertyId
            LIMIT 1
        ];

        if (properties.isEmpty()) {
            return false;
        }

        Property__c propertyRecord = properties[0];

        return (
            propertyRecord.Property_Owner__c == citizenContactId ||
            propertyRecord.Owner__c == citizenContactId
        );
    }

    public static void requireCanAccessProperty(Id propertyId) {

        if (!canAccessProperty(propertyId)) {
            throw new SecurityException(
                'Access denied. This property does not belong to your CivicTax account.'
            );
        }
    }

    // ---------------------------------------------------------
    // ASSESSMENT OWNERSHIP
    // ---------------------------------------------------------

    public static Boolean canAccessAssessment(Id assessmentId) {

        if (assessmentId == null) {
            return false;
        }

        if (isOfficerOrAdmin()) {
            return true;
        }

        Id citizenContactId = getCurrentCitizenContactId();

        List<Tax_Assessment__c> assessments = [
            SELECT Id,
                   Property__c,
                   Property__r.Owner__c,
                   Property__r.Property_Owner__c
            FROM Tax_Assessment__c
            WHERE Id = :assessmentId
            LIMIT 1
        ];

        if (assessments.isEmpty()) {
            return false;
        }

        Tax_Assessment__c assessment = assessments[0];

        if (assessment.Property__c == null) {
            return false;
        }

        return (
            assessment.Property__r.Property_Owner__c == citizenContactId ||
            assessment.Property__r.Owner__c == citizenContactId
        );
    }

    public static void requireCanAccessAssessment(Id assessmentId) {

        if (!canAccessAssessment(assessmentId)) {
            throw new SecurityException(
                'Access denied. This assessment does not belong to your CivicTax account.'
            );
        }
    }

    // OBJECT PERMISSIONS AND FIELD SANITIZATION
    // ---------------------------------------------------------

    public static Boolean canReadObject(String sObjectName) {

        Schema.SObjectType targetType =
            Schema.getGlobalDescribe().get(sObjectName);

        return targetType != null &&
            targetType.getDescribe().isAccessible();
    }

    public static Boolean canCreateObject(String sObjectName) {

        Schema.SObjectType targetType =
            Schema.getGlobalDescribe().get(sObjectName);

        return targetType != null &&
            targetType.getDescribe().isCreateable();
    }

    public static Boolean canUpdateObject(String sObjectName) {

        Schema.SObjectType targetType =
            Schema.getGlobalDescribe().get(sObjectName);

        return targetType != null &&
            targetType.getDescribe().isUpdateable();
    }

    public static List<SObject> sanitizeRecords(
        AccessType accessMode,
        List<SObject> records
    ) {

        if (records == null || records.isEmpty()) {
            return records;
        }

        SObjectAccessDecision decision =
            Security.stripInaccessible(accessMode, records);

        return decision.getRecords();
    }
