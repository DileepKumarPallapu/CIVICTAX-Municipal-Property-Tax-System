import {
  Property,
  PropertyCategory,
  PropertyFilterOptions,
  PropertyVerificationStatus,
} from '../types';
import { IPropertyService } from './interfaces';
import { BACKEND_BASE } from './salesforceApiClient';

interface BackendProperty {
  id: string;
  propertyNumber?: string;
  ownerId?: string;
  ownerName?: string;
  address?: string;
  category?: string;
  builtUpArea?: number;
  zone?: string;
  verificationStatus?: string;
  documentReference?: string;
  latitude?: number;
  longitude?: number;
  constructionYear?: number;
  registrationDate?: string;
}

export class SalesforcePropertyService implements IPropertyService {
  private async request<T>(path: string, options: RequestInit = {}): Promise<T> {
    const response = await fetch(`${BACKEND_BASE}${path}`, {
      ...options,
      credentials: 'include',
      headers: {
        'Content-Type': 'application/json',
        ...options.headers,
      },
    });

    const result = await response.json().catch(() => ({}));

    if (!response.ok || result.success === false) {
      throw new Error(
        result.error || result.message || `CivicTax API error (${response.status})`
      );
    }

    return result as T;
  }

  private mapProperty(record: BackendProperty): Property {
    const category = ['RESIDENTIAL', 'COMMERCIAL', 'INDUSTRIAL'].includes(
      record.category || ''
    )
      ? (record.category as PropertyCategory)
      : 'RESIDENTIAL';

    const validStatuses: PropertyVerificationStatus[] = [
      'DRAFT',
      'SUBMITTED',
      'PENDING',
      'VERIFIED',
      'REJECTED',
      'RESUBMISSION_REQUIRED',
    ];

    const status = validStatuses.includes(
      record.verificationStatus as PropertyVerificationStatus
    )
      ? (record.verificationStatus as PropertyVerificationStatus)
      : 'PENDING';

    const zone = record.zone || '';

    return {
      id: record.id,
      propertyNumber: record.propertyNumber || '',
      ownerId: record.ownerId || '',
      ownerName: record.ownerName || 'Not available',
      ownerEmail: '',
      ownerPhone: '',
      address: record.address || '',
      category,
      zoneId: zone,
      zoneName: zone,
      builtUpArea: Number(record.builtUpArea || 0),
      constructionYear: Number(record.constructionYear || 0),
      documentReference: record.documentReference || '',
      contactDetails: '',
      description: '',
      verificationStatus: status,
      registrationDate: record.registrationDate || '',
      updatedAt: record.registrationDate || '',
      latitude: record.latitude,
      longitude: record.longitude,
      revisions: [],
    };
  }

  public async getProperties(filter?: PropertyFilterOptions): Promise<Property[]> {
    const result = await this.request<{ data: BackendProperty[] }>(
      '/api/properties'
    );

    let properties = (result.data || []).map((record) =>
      this.mapProperty(record)
    );

    if (filter?.search) {
      const query = filter.search.toLowerCase().trim();
      properties = properties.filter((property) =>
        [
          property.propertyNumber,
          property.ownerName,
          property.address,
          property.documentReference,
        ].some((value) => value.toLowerCase().includes(query))
      );
    }

    if (filter?.category && filter.category !== 'ALL') {
      properties = properties.filter(
        (property) => property.category === filter.category
      );
    }

    if (filter?.zoneId && filter.zoneId !== 'ALL') {
      properties = properties.filter(
        (property) => property.zoneId === filter.zoneId
      );
    }

    if (filter?.verificationStatus && filter.verificationStatus !== 'ALL') {
      properties = properties.filter(
        (property) =>
          property.verificationStatus === filter.verificationStatus
      );
    }

    if (filter?.ownerId) {
      properties = properties.filter(
        (property) => property.ownerId === filter.ownerId
      );
    }

    return properties;
  }

  public async getPropertyById(id: string): Promise<Property | null> {
    try {
      const result = await this.request<{ data: BackendProperty }>(
        `/api/properties/${encodeURIComponent(id)}`
      );
      return result.data ? this.mapProperty(result.data) : null;
    } catch (error: any) {
      if (error.message?.includes('(404)')) return null;
      throw error;
    }
  }

  public async getPropertyByNumber(propertyNumber: string): Promise<Property | null> {
    const properties = await this.getProperties();
    return (
      properties.find(
        (property) =>
          property.propertyNumber.toUpperCase() === propertyNumber.toUpperCase()
      ) || null
    );
  }

 
  public async createProperty(
    data: Omit<
      Property,
      | 'id'
      | 'propertyNumber'
      | 'registrationDate'
      | 'updatedAt'
      | 'verificationStatus'
    >
  ): Promise<Property> {
    const result = await this.request<{ data: BackendProperty }>(
      '/api/properties',
      {
        method: 'POST',
        body: JSON.stringify({
          ownerId: data.ownerId,
          ownerName: data.ownerName,
          address: data.address,
          category: data.category,
          zone: data.zoneName || data.zoneId,
          builtUpArea: data.builtUpArea,
          documentReference: data.documentReference,
          constructionYear: data.constructionYear,
        }),
      }
    );

    if (!result.data) {
      throw new Error(
        'Salesforce did not return the registered property.'
      );
    }

    return this.mapProperty(result.data);
  }

  public async updateProperty(
    id: string,
    data: Partial<Property>
  ): Promise<Property> {
    const result = await this.request<{ data: BackendProperty }>(
      `/api/properties/${encodeURIComponent(id)}`,
      {
        method: 'PATCH',
        body: JSON.stringify(data),
      }
    );

    return this.mapProperty(result.data);
  }

  public async verifyProperty(
    id: string,
    status: PropertyVerificationStatus,
    _officerId: string,
    _officerName: string,
    comments?: string
  ): Promise<Property> {
    const result = await this.request<{ data: BackendProperty }>(
      `/api/properties/${encodeURIComponent(id)}/verify`,
      {
        method: 'PATCH',
        body: JSON.stringify({ status, comments }),
      }
    );

    return this.mapProperty(result.data);
  }

  public async deleteProperty(_id: string): Promise<boolean> {
    throw new Error(
      'Salesforce property deletion is disabled. No Salesforce records have been deleted.'
    );
  }
}

import {
  IUserService,
  IPropertyService,
  ITaxRateService,
  IAssessmentService,
  IPaymentService,
  IGrievanceService,
  IReassessmentService,
  INotificationService,
  IDashboardService,
  IZoneService,
} from './interfaces';

import { LocalUserService } from './local/LocalUserService';
import { LocalPropertyService } from './local/LocalPropertyService';
import { SalesforcePropertyService } from './SalesforcePropertyService';
import { LocalTaxRateService } from './local/LocalTaxRateService';
import { LocalAssessmentService } from './local/LocalAssessmentService';
import { LocalPaymentService } from './local/LocalPaymentService';
import { LocalGrievanceService } from './local/LocalGrievanceService';
import { LocalReassessmentService } from './local/LocalReassessmentService';
import { LocalNotificationService } from './local/LocalNotificationService';
import { LocalDashboardService } from './local/LocalDashboardService';
import { LocalZoneService } from './local/LocalZoneService';

class ServiceFactory {
  private static instance: ServiceFactory;

  public readonly userService: IUserService;
  public readonly propertyService: IPropertyService;
  public readonly taxRateService: ITaxRateService;
  public readonly assessmentService: IAssessmentService;
  public readonly paymentService: IPaymentService;
  public readonly grievanceService: IGrievanceService;
  public readonly reassessmentService: IReassessmentService;
  public readonly notificationService: INotificationService;
  public readonly dashboardService: IDashboardService;
  public readonly zoneService: IZoneService;

  private constructor() {
    const useSalesforce = import.meta.env.VITE_USE_SALESFORCE === 'true';

    this.userService = new LocalUserService();
    this.propertyService = useSalesforce
      ? new SalesforcePropertyService()
      : new LocalPropertyService();

    this.taxRateService = new LocalTaxRateService();
    this.assessmentService = new LocalAssessmentService();
    this.paymentService = new LocalPaymentService();
    this.grievanceService = new LocalGrievanceService();
    this.reassessmentService = new LocalReassessmentService();
    this.notificationService = new LocalNotificationService();
    this.dashboardService = new LocalDashboardService();
    this.zoneService = new LocalZoneService();

    console.info(
      `CivicTax Property Data Source: ${useSalesforce ? 'Salesforce' : 'Local Demo'}`
    );
  }

  public static getInstance(): ServiceFactory {
    if (!ServiceFactory.instance) {
      ServiceFactory.instance = new ServiceFactory();
    }
    return ServiceFactory.instance;
  }
}

export const services = ServiceFactory.getInstance();

import { IProperty, ITaxAssessment, ITaxPayment, IGrievance, IReassessment, IDashboardMetrics } from './types.js';

export interface ICivicTaxRepository {
  // Properties
  listProperties(filter?: { category?: string; zone?: string; status?: string }): Promise<IProperty[]>;
  getPropertyById(id: string): Promise<IProperty | null>;
  createProperty(property: Omit<IProperty, 'id' | 'registrationDate'>): Promise<IProperty>;
  updatePropertyStatus(id: string, status: IProperty['verificationStatus']): Promise<IProperty>;

  // Assessments
  listAssessments(propertyId?: string): Promise<ITaxAssessment[]>;
  getAssessmentById(id: string): Promise<ITaxAssessment | null>;
  createAssessment(assessment: Omit<ITaxAssessment, 'id'>): Promise<ITaxAssessment>;

  // Payments
  listPayments(assessmentId?: string): Promise<ITaxPayment[]>;
  createPayment(payment: Omit<ITaxPayment, 'id' | 'receiptNumber'>): Promise<{ payment: ITaxPayment; assessment: ITaxAssessment }>;

  // Grievances
  listGrievances(citizenId?: string): Promise<IGrievance[]>;
  createGrievance(grievance: Omit<IGrievance, 'id' | 'grievanceNumber' | 'createdAt'>): Promise<IGrievance>;
  resolveGrievance(id: string, comments: string): Promise<IGrievance>;

  // Reassessments
  listReassessments(propertyId?: string): Promise<IReassessment[]>;
  createReassessment(reassessment: Omit<IReassessment, 'id' | 'requestNumber' | 'createdAt'>): Promise<IReassessment>;

  // Dashboard Metrics
  getDashboardMetrics(): Promise<IDashboardMetrics>;
}

import { SalesforceSessionData } from '../types/index.js';

export class BackendServiceFactory {
  public static getRepository(
    session?: SalesforceSessionData
  ): ICivicTaxRepository {
    if (session?.accessToken && session?.instanceUrl) {
      return new SalesforceRepository(session);
    }

    return new LocalRepository();
  }

import { Router, Request, Response } from 'express';
import { BackendServiceFactory } from '../repositories/ServiceFactory.js';
import { config } from '../config/index.js';

export const municipalApiRouter = Router();

const getRequestRepository = (req: Request) =>

// 1. HEALTH & SYSTEM STATUS

// GET /api/health
municipalApiRouter.get('/api/health', (req: Request, res: Response) => {

    service: 'CivicTax Municipal Backend Gateway',
    timestamp: new Date().toISOString(),
    dataProvider: 'LocalRepository (Authoritative In-Memory Schema)',
  });
});

// GET /api/salesforce/health
municipalApiRouter.get('/api/salesforce/health', (req: Request, res: Response) => {
  const sfSession = req.session.salesforceSession;

  res.json({
    status: 'HEALTHY',
    service: 'CivicTax Salesforce Gateway',
    salesforceConnected: Boolean(sfSession && sfSession.accessToken),
    apiVersion: config.salesforce.apiVersion,
    loginUrl: config.salesforce.loginUrl,
    orgId: sfSession ? sfSession.orgId : null,
    timestamp: new Date().toISOString(),
  });
});

// GET /api/properties
municipalApiRouter.get('/api/properties', async (req: Request, res: Response) => {
  try {
    const { category, zone, status } = req.query;

    const properties = await getRequestRepository(req).listProperties({
      category: category as string,
      zone: zone as string,
      status: status as any,
    });

    res.json({
      success: true,
      count: properties.length,
      data: properties,
    });
  } catch (err: any) {
    res.status(500).json({
      success: false,
      error: err.message,
    });
  }
});

// POST /api/properties
municipalApiRouter.post('/api/properties', async (req: Request, res: Response) => {
  try {
    const {
      propertyNumber,
      address,
      category,
      builtUpArea,
      zone,
      documentReference,
      constructionYear,
      latitude,
      longitude,
      ownerId,
      ownerName,
    } = req.body;

    // Validate mandatory registration fields.
    // propertyNumber is optional because the backend generates it.
    if (!address || !category || builtUpArea === undefined || builtUpArea === null || builtUpArea === '') {
      res.status(400).json({
        success: false,
        error: 'Missing mandatory property registration fields',
      });
      return;
    }

    const area = Number(builtUpArea);

    if (!Number.isFinite(area) || area <= 0) {
      res.status(400).json({
        success: false,
        error: 'Built-up area must be strictly greater than zero',
      });
      return;
    }

    const generatedPropertyNumber =
      propertyNumber ||
      `CT-${new Date().getFullYear()}-${Date.now()}`;

    const created = await getRequestRepository(req).createProperty({
      propertyNumber: generatedPropertyNumber,
      ownerId: ownerId || 'usr-citizen-01',
      ownerName: ownerName || 'Rajesh Kumar Sharma',
      address: String(address).trim(),
      category,
      builtUpArea: area,
      zone: zone || 'Z-CENTRAL',
      verificationStatus: 'PENDING',
      documentReference: documentReference || 'DOC-PENDING',
      constructionYear: constructionYear
        ? Number(constructionYear)
        : undefined,
      latitude: latitude !== undefined ? Number(latitude) : undefined,
      longitude: longitude !== undefined ? Number(longitude) : undefined,
    });

    res.status(201).json({
      success: true,
      message: 'Property registered successfully',
      data: created,
    });
  } catch (err: any) {
    res.status(400).json({
      success: false,
      error: err.message,
    });
  }
});

// GET /api/properties/:id
municipalApiRouter.get('/api/properties/:id', async (req: Request, res: Response) => {
  try {
    const prop = await getRequestRepository(req).getPropertyById(req.params.id);

    if (!prop) {
      res.status(404).json({
        success: false,
        error: 'Property not found',
      });
      return;
    }

    res.json({
      success: true,
      data: prop,
    });
  } catch (err: any) {
    res.status(500).json({
      success: false,
      error: err.message,
    });
  }
});

authRouter.get('/auth/salesforce/callback', async (req: Request, res: Response) => {
  const { code, state, error, error_description } = req.query;

  if (error) {
    res.redirect(`${config.frontendUrl}/admin/salesforce?error=${encodeURIComponent(String(error_description || error))}`);
    return;
  }

  if (!code || !state) {
    res.redirect(`${config.frontendUrl}/admin/salesforce?error=Missing_Code_Or_State`);
    return;
  }

  // Validate state against CSRF
  if (state !== req.session.oauthState) {
    res.redirect(`${config.frontendUrl}/admin/salesforce?error=Invalid_OAuth_State`);
    return;
  }

  const verifier = req.session.codeVerifier;
  if (!verifier) {
    res.redirect(`${config.frontendUrl}/admin/salesforce?error=Missing_Code_Verifier`);
    return;
  }

  try {
    const tokenData = await SalesforceClient.exchangeCodeForToken(String(code), verifier);

    // Fetch user identity
    let username = '';
    let displayName = '';
    let orgId = '';
    let userId = '';

    try {
      const identity = await SalesforceClient.fetchIdentity(tokenData.id, tokenData.access_token);
      username = identity.username;
      displayName = identity.display_name;
      orgId = identity.organization_id;
      userId = identity.user_id;
    } catch {
      // Identity URL parsing fallback
      const segments = tokenData.id.split('/');
      orgId = segments[segments.length - 2] || '';
      userId = segments[segments.length - 1] || '';
    }

    const sessionData: SalesforceSessionData = {
      accessToken: tokenData.access_token,
      refreshToken: tokenData.refresh_token,
      instanceUrl: tokenData.instance_url,
      identityUrl: tokenData.id,
      userId,
      orgId,
      username,
      displayName,
      connectedAt: new Date().toISOString(),
      lastActiveAt: new Date().toISOString(),
    };

    req.session.salesforceSession = sessionData;
    delete req.session.oauthState;
    delete req.session.codeVerifier;

    res.redirect(`${config.frontendUrl}/admin/salesforce?status=success`);
  } catch (err: any) {
    console.error('Salesforce OAuth Error:', err);
    res.redirect(`${config.frontendUrl}/admin/salesforce?error=${encodeURIComponent(err.message || 'Authentication_Failed')}`);
  }
});
